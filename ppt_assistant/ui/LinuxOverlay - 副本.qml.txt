    // Update mask for click-through
    function updateMask() {
        var rects = []

        // Status bar
        if (showStatusBar && statusBar.visible) {
            rects.push({
                x: statusBar.x,
                y: statusBar.y,
                width: statusBar.width,
                height: statusBar.height,
                role: "status-bar"
            })
        }

        // Flippers
        if (leftFlipper.visible) {
            rects.push({
                x: leftFlipper.x,
                y: leftFlipper.y,
                width: leftFlipper.width,
                height: leftFlipper.height,
                role: "flipper"
            })
        }
        if (rightFlipper.visible) {
            rects.push({
                x: rightFlipper.x,
                y: rightFlipper.y,
                width: rightFlipper.width,
                height: rightFlipper.height,
                role: "flipper"
            })
        }

        // Toolbar
        if (toolbarContainer.visible && toolbar.visible) {
            rects.push({
                x: toolbarContainer.x + toolbar.x,
                y: toolbarContainer.y + toolbar.y,
                width: toolbar.width,
                height: toolbar.height,
                role: "toolbar"
            })
        }

        // Page selector
        if (pageSelectorVisible) {
            rects.push({
                x: pageSelector.x,
                y: pageSelector.y,
                width: pageSelector.width,
                height: pageSelector.height,
                role: "page-selector"
            })
        }

        // Popups
        if (colorPickerVisible) {
            rects.push({
                x: colorPicker.x,
                y: colorPicker.y,
                width: colorPicker.width,
                height: colorPicker.height,
                role: "popup"
            })
        }
        if (eraserPopupVisible) {
            rects.push({
                x: eraserPopup.x,
                y: eraserPopup.y,
                width: eraserPopup.width,
                height: eraserPopup.height,
                role: "popup"
            })
        }

        if (bridge) {
            bridge.updateMask(rects)
        }
    }
